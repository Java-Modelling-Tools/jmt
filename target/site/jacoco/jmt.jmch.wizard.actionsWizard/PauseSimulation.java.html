 package jmt.jmch.wizard.actionsWizard;

import java.awt.event.ActionEvent;

import jmt.jmch.wizard.panels.JMCHWizardPanel;

/**
 * @author Lorenzo Torri
 * Date: 30-mar-2024
 * Time: 15.21
 */
public class PauseSimulation extends AbstractMCHAction {

    private static final long serialVersionUID = 1L;

    /**
     * Defines an <code>Action</code> object with a default
     * description string and default icon.
     */
    public PauseSimulation(JMCHWizardPanel panel) {
        super("Pause Simulation", "Pause", panel);
        putValue(SHORT_DESCRIPTION, "Pause simulation");
        //putValue(MNEMONIC_KEY, new Integer(KeyEvent.VK_S));
    }

    /**
     * Invoked when an action occurs.
     */
    public void actionPerformed(ActionEvent e) {
        panel.pauseAnimation();
    }

}
