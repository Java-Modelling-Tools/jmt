	public DPoint calcEquiUtilizationPoint(Vector<newPoint> vertices2D) {
		// NEW Spicuglia
		// If there are two station we calculate the equiutilization point
		if (vertices2D.size() == 2) {
			double L11, L12, L21, L22;
			double b1;
			double b2;
	
			L11 =  vertices2D.get(0).x;
			L12 =  vertices2D.get(0).y;
			L21 =  vertices2D.get(1).x;
			L22 = vertices2D.get(1).y;
			b1 = Math.log(L22/L12) / Math.log((L11*L22)/(L12*L21));
			b2 = 1 - b1;
			return new DPoint(b1, b2);
		}
		return null;
		// END Spicuglia
	}
