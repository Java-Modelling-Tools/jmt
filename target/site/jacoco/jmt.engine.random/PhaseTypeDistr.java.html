package jmt.engine.random;

import jmt.common.exception.IncorrectDistributionParameterException;

/**
 * 
 * This is the Phase-Type distribution.
 * 
 * <br><br>Copyright (c) 2017
 * <br>Imperial Collge London - Department of Computing
 * @author Lulai Zhu
 * 
 */
public class PhaseTypeDistr extends AbstractDistribution implements Distribution {

	private int curState;
	private Exponential expDistr;
	private ExponentialPar expPar;

	/**
	 * This is the constructor. It creates a new phase-type distribution which
	 * is constituted by arbitrary exponential phases.
	 *
	 */
	public PhaseTypeDistr() {
		curState = -1;
		expDistr = new Exponential();
		expPar = new ExponentialPar();
	}

	/**
	 * This method is used to obtain from the distribution its probability distribution
	 * function evaluated where required by the user.
	 *
	 * @param x double indicating where to evaluate the pdf.
	 * @param p parameter of the phase-type distribution.
	 * @throws IncorrectDistributionParameterException
	 * @return double with the probability distribution function evaluated in x.
	 */
	public double pdf(double x, Parameter p) throws IncorrectDistributionParameterException {
		return 0.0;
	}

	/**
	 * This method is used to obtain from the distribution its cumulative distribution
	 * function evaluated where required by the user.
	 *
	 * @param x double indicating where to evaluate the cdf.
	 * @param p parameter of the phase-type distribution.
	 * @throws IncorrectDistributionParameterException
	 * @return double with the cumulative distribution function evaluated in x.
	 */
	public double cdf(double x, Parameter p) throws IncorrectDistributionParameterException {
		return 0.0;
	}

	/**
	 * This method is used to obtain from the distribution the value of its own
	 * theoretic mean.
	 *
	 * @param p parameter of the phase-type distribution.
	 * @throws IncorrectDistributionParameterException
	 * @return double with the theoretic mean of the distribution.
	 */
	public double theorMean(Parameter p) throws IncorrectDistributionParameterException {
		return ((PhaseTypePar) p).getMean();
	}

	/**
	 * This method is used to obtain from the distribution the value of its own
	 * theoretic variance.
	 *
	 * @param p parameter of the phase-type distribution.
	 * @throws IncorrectDistributionParameterException
	 * @return double with the theoretic variance of the distribution.
	 */
	public double theorVariance(Parameter p) throws IncorrectDistributionParameterException {
		return ((PhaseTypePar) p).getVar();
	}

	/**
	 * This method is used to obtain from the distribution the next number distributed
	 * according to the distribution parameter.
	 *
	 * @param p parameter of the phase-type distribution.
	 * @throws IncorrectDistributionParameterException
	 * @return double with the next random number of this distribution.
	 */
	public double nextRand(Parameter p) throws IncorrectDistributionParameterException {
		double sample = 0.0;
		double[] alpha = ((PhaseTypePar) p).getAlpha();
		double[] lambda = ((PhaseTypePar) p).getLambda();
		double[][] P = ((PhaseTypePar) p).getP();

		double rand = engine.raw();
		double sum = 0.0;
		for (int i = 0; i < alpha.length; i++) {
			sum += alpha[i];
			if (sum >= rand) {
				curState = i;
				break;
			}
		}

		if (expDistr.engine == null) {
			expDistr.setRandomEngine(engine);
		}
		
		OUTER_LOOP:
		while (true) {
			expPar.setLambda(lambda[curState]);
			sample += expDistr.nextRand(expPar);

			rand = engine.raw();
			sum = 0.0;
			for (int i = 0; i < alpha.length; i++) {
				sum += P[curState][i];
				if (sum >= rand) {
					curState = i;
					continue OUTER_LOOP;
				}
			}
			break;
		}

		return sample;
	}

} // end PhaseTypeDistr
