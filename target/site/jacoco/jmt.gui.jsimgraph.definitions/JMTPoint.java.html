package jmt.gui.jsimgraph.definitions;

import java.awt.geom.Point2D;

/**
 * <p><b>Name:</b> JMTPoint</p> 
 * <p><b>Description:</b> 
 * A Point2D implementation class used to store position of a station 
 * </p>
 * <p><b>Date:</b> 21/feb/07
 * <b>Time:</b> 18:37:35</p>
 * @author Bertoli Marco
 * @version 1.0
 */
public class JMTPoint extends Point2D {

	private double x;
	private double y;
	private boolean rotate;
	private double angle;

	/**
	 * Builds a new JMTPoint at position (0,0) and no rotation.
	 */
	public JMTPoint() {
		x = 0;
		y = 0;
		rotate = false;
		angle = 0;
	}

	/**
	 * Builds a new JMTPoint
	 * @param x x coordinate
	 * @param y y coordinate
	 * @param rotate true if station is rotated
	 */
	public JMTPoint(double x, double y, boolean rotate) {
		this.x = x;
		this.y = y;
		this.rotate = rotate;
		this.angle = 0;
	}

	/**
	 * Builds a new JMTPoint from a given point
	 * @param x x coordinate
	 * @param y y coordinate
	 * @param rotate true if station is rotated
	 * @param angle value of the rotation angle
	 */
	public JMTPoint(double x, double y, boolean rotate, double angle) {
		this.x = x;
		this.y = y;
		this.rotate = rotate;
		this.angle = angle;
	}

	/**
	 * Builds a new JMTPoint from a given point
	 * @param point point for coordinates
	 * @param rotate true if station is rotated
	 */
	public JMTPoint(Point2D point, boolean rotate) {
		this.x = point.getX();
		this.y = point.getY();
		this.rotate = rotate;
		this.angle = 0;
	}

	/**
	 * Builds a new JMTPoint from a given point
	 * @param point point for coordinates
	 * @param rotate true if station is rotated
	 * @param angle value of the rotation angle
	 */
	public JMTPoint(Point2D point, boolean rotate, double angle) {
		this.x = point.getX();
		this.y = point.getY();
		this.rotate = rotate;
		this.angle = angle;
	}

	/* (non-Javadoc)
	 * @see java.awt.geom.Point2D#getX()
	 */
	@Override
	public double getX() {
		return x;
	}

	/* (non-Javadoc)
	 * @see java.awt.geom.Point2D#getY()
	 */
	@Override
	public double getY() {
		return y;
	}

	/* (non-Javadoc)
	 * @see java.awt.geom.Point2D#setLocation(double, double)
	 */
	@Override
	public void setLocation(double x, double y) {
		this.x = x;
		this.y = y;
	}

	/* (non-Javadoc)
	 * @see java.awt.geom.Point2D#clone()
	 */
	@Override
	public Object clone() {
		return new JMTPoint(x, y, rotate);
	}

	/**
	 * @return true if station is rotated
	 */
	public boolean isRotate() {
		return rotate;
	}

	/**
	 * @return the value of the rotation angle
	 */
	public double getAngle() {
		return angle;
	}

	/**
	 * @param angle = value of the rotation angle
	 */
	public void setAngle(double angle) {
		this.angle= angle;
	}

	/**
	 * @param rotate true if station is rotated
	 */
	public void setRotate(boolean rotate) {
		this.rotate = rotate;
	}

	/**
	 * @param x x coordinate
	 */
	public void setX(double x) {
		this.x = x;
	}

	/**
	 * @param y y coordinate
	 */
	public void setY(double y) {
		this.y = y;
	}

}
