package jmt.jmarkov.Queues.Exceptions;

/**
 * MMQueues
 * --------------------------------------
 * 20-mar-2004 - Queues.Exceptions/NoJobsException.java
 * 
 * @author Ernesto
 */
public class NoJobsException extends Exception {

	/**
	 * 
	 */
	private static final long serialVersionUID = 1L;

	/**
	 * 
	 */
	public NoJobsException() {
		super();
	}

	/**
	 * @param message
	 */
	public NoJobsException(String message) {
		super(message);
	}

	/**
	 * @param cause
	 */
	public NoJobsException(Throwable cause) {
		super(cause);
	}

	/**
	 * @param message
	 * @param cause
	 */
	public NoJobsException(String message, Throwable cause) {
		super(message, cause);
	}

}
